  # ===== 半角の括弧内を除去する。単純な装備ラベル用で、入れ子の括弧の解析はしない =====

        # 分割結果に括弧後の要素がない場合は追加しない。閉じ括弧のない断片も捨てる。
      end
    end
    ret ($result | string.strip)
  end

  func extract_weapon

      ret ""

    $weapon = $apparel_parts[0] | string.replace "|" "" | string.strip

    if $weapon == ""
      ret ""

    ret "[Weapon] " + $weapon

  # 現在の作業対象・手荷物・台に置いた材料から遺体だけを拾う。名前や作業名では判定しない。

  func format_job_corpse_notes

    if $pawn == null || $pawn.CurJob == null
      ret ""

    $job = $pawn.CurJob

    # 未着手のジョブや材料キューは走査せず、消費済み（Countが0）の材料も除外する。
    if $job.placedThings != null
      for $placed in $job.placedThings
        if $placed != null && $placed.Count > 0
          $targets[$targets.size] = $placed.thing
        end
      end
    end

    $seen_ids = []
    $notes = []
    for $target in $targets
      if $target == null || $target.Destroyed || $target.def == null || !$target.def.IsCorpse
        continue
      end
      $dead_pawn = $target.InnerPawn
      if $dead_pawn == null || ($seen_ids | array.contains $target.thingIDNumber)
        continue
      end
      # 同じ遺体が作業対象と手荷物などに重複しても、一度だけ表示する。
      $seen_ids[$seen_ids.size] = $target.thingIDNumber

      # 判定基準はプレイヤー陣営。囚人・奴隷を仲間と取り違えず、非敵対も同盟とは断定しない。
      # 死亡時の外交関係ではなく、遺体のInnerPawnに残る所属と現在の関係を参照する。
      $faction_name = format_faction_name $dead_pawn.faction $faction_dict
      if $dead_pawn.IsPrisoner
        $standing = "囚人"
      else if $dead_pawn.IsSlave
        $standing = "奴隷"
      else if $dead_pawn.IsColonist || ($dead_pawn.HomeFaction != null && $dead_pawn.HomeFaction.IsPlayer)
        $standing = "プレイヤー陣営の仲間"
      else if IsEnemy $dead_pawn
        $standing = "プレイヤー陣営に敵対"
      else if $faction_name != ""
        $standing = "プレイヤー陣営には非敵対"
      else
        $standing = "所属派閥なし"
      end
      if $faction_name != ""
        $standing = $standing + "／所属派閥「" + $faction_name + "」"
      end

      $dead_name = $dead_pawn.name
      if $dead_name == null || $dead_name == ""
        $dead_name = "名前不明のポーン"
      end

      # Humanlike/Animalより異常存在を優先する。IsMutantで人間由来のグールなども区別する。
      # 派閥名では分類しないため、味方になった異常存在や改名された派閥でも種別を保てる。
      $corpse_kind = ""
      $race = $dead_pawn.RaceProps
      if $race != null
        if $race.IsAnomalyEntity || $dead_pawn.IsEntity || $dead_pawn.IsMutant
          $corpse_kind = "異常存在"
        else if $race.IsMechanoid
          $corpse_kind = "メカノイド"
        else if $race.Humanlike
          # HumanlikeにはMODの人型種族も含まれるため、Human以外は種族の表示名を使う。
          if $dead_pawn.def != null && $dead_pawn.def.defName == "Human"
            $corpse_kind = "人間"
          else if $dead_pawn.def != null && $dead_pawn.def.label != null && $dead_pawn.def.label != ""
            $corpse_kind = $dead_pawn.def.label
          else
            $corpse_kind = "人型種族"
          end
        else if $race.Animal
          $corpse_kind = "動物"
        end
      end
      # 判別できない場合は種別を推測せず、従来どおり名前と遺体だけを表示する。
      if $corpse_kind != ""
        $dead_name = $dead_name + "という" + $corpse_kind
      end
      $notes[$notes.size] = "作業対象の補足：" + $dead_name + "の遺体（" + $standing + "）"
    end

    ret $notes | array.join "\n"
  end

  # ===== 場所情報を一定の形式に整え、天候・外部環境を補う =====
  # 温度は35℃以上または-10℃以下の場合だけ表示し、軌道上では表示しない
  func format_location
    $text = $0
    $weather_text = $1
    $map_event_text = $2

    if $text == null || $text == ""
      $environment_result = ""
      if $weather_text != null && $weather_text != ""
        $environment_result = "天候・外部環境：" + $weather_text
      end
      if $map_event_text != null && $map_event_text != ""
        if $environment_result != ""
          $environment_result = $environment_result + " / "
        end
        $environment_result = $environment_result + "マップイベント：" + $map_event_text
      end
      ret $environment_result
    end

    $parts = $text | string.split ";"
    $location_type = $parts[0] | string.strip
    $result = ""
    $is_orbital = $weather_text != null && ($weather_text | string.contains "軌道上")

    if $location_type | string.contains "室内"
      $result = "区分：室内"
    else if $location_type | string.contains "外気"
      $result = "区分：屋外"
    else
      $result = "区分：" + $location_type
    end

    # p.locationの3要素目は部屋名
    if $parts.size > 2
      $room_name = $parts[2] | string.strip
      if $room_name != ""
        $result = $result + " / 部屋：" + $room_name
      end
    end

    # p.locationの2要素目はpawn位置の実温度。会話に影響しやすい極端な時だけ残す
    if $parts.size > 1 && !$is_orbital
      $temperature_text = $parts[1] | string.strip
      if $temperature_text != ""
        $temperature_number_text = $temperature_text | string.replace "℃" "" | string.replace "°" "" | string.replace "C" "" | string.replace "c" "" | string.strip
        $temperature_value = $temperature_number_text | string.to_float
        if $temperature_value != null && ($temperature_value >= 35 || $temperature_value <= -10)
          $result = $result + " / 温度：" + $temperature_text
        end
      end
    end

    if $weather_text != null && $weather_text != ""
      $result = $result + " / 天候・外部環境：" + $weather_text
    end

    if $map_event_text != null && $map_event_text != ""
      $result = $result + " / マップイベント：" + $map_event_text
    end

    ret $result
  end

  # ===== カンマ区切り文字列の後処理（空要素・余分な空白の除去） =====
  func clean_list
    $text = $0
    $parts = $text | string.split ","
    $result = ""
    for $part in $parts
      $trimmed = $part | string.strip
      if $trimmed != ""
        if $result != ""
          $result = $result + ", "
        end
        $result = $result + $trimmed
      end
    end
    ret $result
  end
  # ===== 関係情報を現在の会話に関係する人物だけに絞る（配偶者なども含む） =====
  # 例:
  #   "Relations: リンダ(妻), カーミラ(姉)" → "リンダ(妻), カーミラ(姉)"
  #
  # 以下のどちらかを満たす場合だけ残す
  # ・名前が pawns 内に存在する
  # ・名前が現在の p.job に含まれる
  func filter_relations
    $text = $0
    $pawn_list = $1
    $job_text = $2

    if $text == null || $text == ""
      ret ""
    end

    # 接頭辞を残すと「Relations: リンダ」とpawn名「リンダ」が一致しないため、比較前に除く。
    $relation_text = $text | string.replace "Relations:" "" | string.strip
    $parts = $relation_text | string.split ","
    $result = ""

    for $part in $parts
      $relation = $part | string.strip

      if $relation != ""

        # "リンダ(妻)" → "リンダ"
        $name_parts = $relation | string.split "("
        $relation_name = $name_parts[0] | string.strip

        $keep = false

        # ===== 現在会話中のpawnにいるか =====
        for $other_p in $pawn_list
          if $other_p.name && $relation_name == ($other_p.name | string.strip)
            $keep = true
            break
          end
        end

        # ===== 現在のjobに名前が含まれているか =====
        if !$keep && $job_text && ($job_text | string.contains $relation_name)
          $keep = true
        end

        # ===== 条件を満たした関係だけ追加 =====
        if $keep
          if $result != ""
            $result = $result + ", "
          end
          $result = $result + $relation
        end

      end
    end

    ret $result
  end

  # ===== 部分置換（内部利用）。置換後にclean_listで整形 =====
  # keyが重複・包含関係にある場合は登録順で結果が変わるため、具体的な長いkeyを先に置く
  # 空のkeyはstring.replaceの例外になるため無視する。
  func partial_replace
    $text = $0
    $dict = $1
    if $text == null || $dict == null
      ret $text
    end
    $result = $text
    for $key in ($dict | object.keys)
      if $key != ""
        $keyword = $dict[$key]
        $result = $result | string.replace $key $keyword
      end
    end
    ret (clean_list $result)
  end

  # ===== メインの置換窓口 =====
  # modeが"partial"の時だけ部分置換する。それ以外、または値・辞書がnullならそのまま返す。
  func apply_replace
    $value = $0
    $dict = $1
    $mode = $2
    if $value == null || $dict == null
      ret $value
    end
    if $mode == "partial"
      ret (partial_replace $value $dict)
    end
    ret $value
  end

  # ===== 健康情報専用の置換 =====
  # valueが空文字: keyを含むHediff項目全体を除外
  # valueが空以外: keyに一致した部分だけをvalueへ置換
  # 空のkeyは全項目一致や置換例外を避けるため無視する。
  func replace_health_label
    $health_text = $0
    $health_dict = $1

    if $health_text == null || $health_text == "" || $health_dict == null
      ret $health_text
    end

    # simple function内の通常変数は呼び出し元と共有されるため、$付きのローカル変数を使う
    $health_result = $health_text
    for $health_key in ($health_dict | object.keys)
      if $health_key != "" && ($health_result | string.contains $health_key)
        $keyword = $health_dict[$health_key]
        if $keyword == ""
          ret ""
        end
        $health_result = $health_result | string.replace $health_key $keyword
      end
    end

    ret ($health_result | string.strip)
  end

  # ===== RimTalkの健康情報から出血状態を抜き出す =====
  # 例: "Health: Bleeding (severe, death in 19.2 hours), ..."
  #  → "Bleeding (severe, death in 19.2 hours)"
  func extract_bleeding_status
    $text = $0
    $marker = "Bleeding ("

    if $text == null || $text == "" || !($text | string.contains $marker)
      ret ""
    end

    # string.splitは空要素を捨てるため、冒頭がBleedingでも先頭の非対象要素を残す。
    $parts = (" " + $text) | string.split $marker
    for $part in $parts
      if for.index > 0
        $detail_parts = $part | string.split ")"
        if $detail_parts.size > 0 && $detail_parts[0] != ""
          ret $marker + $detail_parts[0] + ")"
        end
      end
    end

    ret ""
  end

  # ===== 健康情報本文と、採用したHediffの置換前ラベル一覧をまとめて返す =====
  # GetHediffsの結果を時間・除去辞書で絞る。source_labelsは攻撃元の補足専用で、非表示項目を含めない。
  func format_health_conditions

    $threshold_ticks = $1
    $current_game_ticks = $2
    $health_text = $pawn.health
    $result = ""
    $source_labels = []

    if $health_text == null || $health_text == ""
      ret {text: "", source_labels: []}

      if $hediff.ageTicks < $current_game_ticks && $hediff.ageTicks <= $threshold_ticks
        # 空のvalueは項目全体の除外、それ以外は一致部分の置換に使う
        $health_label = replace_health_label $hediff.LabelCap replace_dict

        if $health_label != null && $health_label != ""
          $source_labels[$source_labels.size] = $hediff.LabelCap
          $hours_ago = ($hediff.ageTicks / 2500) | math.round 1

  # ===== ラベル+値の1ブロックを作る。値がnull/空文字ならスキップ =====
  func add_line

  # 表示フラグ: falseなら非表示。trueまたはnullなら、値がある場合に表示する（危険時・狩猟時など）。

  func prepare_field

  # ===== 完全一致辞書に従って文字列を確率表示する =====
  # 辞書にない文字列は重要な可能性があるため、そのまま表示する
  func filter_by_exact_chance
    $source_text = $0
    $chance_dict = $1

    if $source_text == null || $source_text == ""
      ret ""
    end

    $trimmed = $source_text | string.strip
    $chance = null
    if $chance_dict != null
      $chance = $chance_dict[$trimmed]
    end

    if $chance == null || $chance >= 100
      ret $trimmed
    end
    if $chance <= 0
      ret ""
    end

    if (random 0 100) < $chance
      ret $trimmed
    end

    ret ""
  end

  # ===== 現在のマップで発生中のGameConditionを取得する =====
  # 性別を絞り、完全な表示名→本体名の順で確率辞書を引く。抽選は呼び出しごと（pawnごと）に行う。
  func format_map_events

    $chance_dict = $1
    $pawn_gender = $2

    $result = ""

            $male_scan_label = $gender_scan_label | string.replace "female" ""
            $targets_male = ($male_scan_label | string.contains "男性") || ($male_scan_label | string.contains "male")
            $gender_matches = (!$targets_female && !$targets_male) || ($targets_female && $pawn_gender == "Female") || ($targets_male && $pawn_gender == "Male")

            end
          end
        end
      end
    end

    ret $result
  end

  # ===== 欲求を表示確率に従って抽出 =====
  func filter_thoughts
    $text = $0
    $dict = $1

    if $text == null || $text == "" || $dict == null
      ret ""
    end

    $parts = $text | string.split ","
    $result = ""

    for $part in $parts
      $trimmed = $part | string.strip
      $show = false

      for $key in ($dict | object.keys)
        if $trimmed != "" && $key != "" && ($trimmed | string.contains $key)

          $chance = $dict[$key]

          if $chance >= 100
            $show = true
          else if $chance > 0 && (random 0 100) < $chance
            $show = true
          end

          break
        end
      end

      if $show
        if $result != ""
          $result = $result + ", "
        end
        $result = $result + $trimmed
      end
    end

    ret $result
  end

  # ===== personalityに特殊文字列が含まれていた場合の処理 =====
  # text: p.personalityそのもの
  # context_text: {"キーワード"}判定用の人物情報。UN発火時だけ、取得・置換済みシーン文も含める。
  # danger_flag: {!}判定に使うtrue/false
  # UN_text: fullinteractionまたはp.jobから取得・置換したUN文字列。{UN}判定にも使う。
  # 制約: 特殊構文は1行につき1個とし、本文中には追加の「}」を含めない
  # 通常行に波括弧を使うと特殊構文と解釈されるため、人物設定では避ける
  func check_personality
    $text = $0
    $context_text = $1
    $danger = $2
    $UN_text = $3

    $split_text = $text | string.split "\n"
    $result = ""

    for $line in $split_text

      if ($line | string.contains "{") && ($line | string.contains "}")

        # ---- 1. 全ての"}"で分割し、先頭2要素をタグと本文として使う ----
        # 例: {20%}実は趣味がある{END}
        #  → parts[0] = "{20%"
        #  → parts[1] = "実は趣味がある{END"
        $parts = $line | string.split "}"

        if $parts.size >= 2

          $tag_part  = $parts[0]
          $body_part = $parts[1]

          # ---- 2. タグの中身だけ取り出す（先頭の "{" を削除） ----
          $tag = $tag_part | string.replace "{" ""

          # ---- 3. 本文だけ取り出す（末尾の "{END" を削除） ----
          $body = $body_part | string.replace "{END" ""

          $show_line = false

          if $tag | string.contains "%"
            # ===== パターンA: {20%}あいうえお{END} =====
            # "20%" → "20" にしてから数値に変換
            $percent_num = ($tag | string.replace "%" "") | string.to_int

            if $percent_num >= 100 || ($percent_num > 0 && (random 0 100) < $percent_num)
              $show_line = true
            end

          else
            if $tag == "!"
              # ===== パターンB: {!}あいうえお{END} =====
              if $danger
                $show_line = true
              end
            else if $tag == "UN"
              # ===== パターンC: {UN}あいうえお{END} =====
              # UNが発火し、fullinteractionまたはp.jobからシーン文を取得できた場合だけ表示する。
              if $UN_text != null && $UN_text != ""
                $show_line = true
              end
            else
              # ===== パターンD: {"キーワード"}あいうえお{END} =====
              $keyword = $tag | string.replace "\"" ""
              if $context_text && ($context_text | string.contains $keyword)
                $show_line = true
              end
            end
          end

          if $show_line
            if $result != ""
              $result = $result + "\n"
            end
            $result = $result + $body
          end

        end

      else
        # 特殊記法が含まれていない普通の行はそのまま出力
        if $result != ""
          $result = $result + "\n"
        end
        $result = $result + $line
      end

    end

    ret $result
  end
