  # ===== 特殊シーン切り替え設定（System側と必ず同じ値にする） =====
  # 両テンプレートは別々に評価される。設定が食い違うと、SystemとUserで対象人数が不一致になる
  # key: fullinteractionから検索する文字列
  # value: 空なら一致行をそのまま使用し、値があれば一致行全体を置換する
  NS_keyword = {
    "キャッチボール": "",
    "球技をして遊んだ": "ボールで遊んでいた",
  }

  # "Male" / "Female" / "ALL"（性別を問わない）
  NS_gender = "Male"

  func find_interaction_line
    $interaction_text = $0
    $keyword_dict = $1

    if $interaction_text == null || $interaction_text == "" || $keyword_dict == null
      ret ""
    end

    # fullinteractionの行順を優先する。同じ行に複数keyがある場合は辞書の登録順で最初のものを使う
    # valueが空なら原文行、空でなければvalueだけを返す（部分置換ではなく行全体の置換）
    for $line in ($interaction_text | SplitByLine)
      $trimmed = $line | string.strip
      if $trimmed != ""
        for $key in ($keyword_dict | object.keys)
          if $key != "" && ($trimmed | string.contains $key)
            $replacement = $keyword_dict[$key]
            if $replacement == null || $replacement == ""
              ret $trimmed
            end
            ret $replacement
          end
        end
      end
    end

    ret ""
  end

  func find_NS_pawn
    $pawn_list = $0
    $target_gender = $1
    $keyword_dict = $2

    if $target_gender == "" || $keyword_dict == null
      ret null
    end

    $keys = $keyword_dict | object.keys
    if $keys.size == 0
      ret null
    end

    # 複数pawnが一致しても最初の1人だけを採用し、会話をNS用の単独出力へ切り替える
    # ALLは性別条件だけを無効化し、キーワード一致は引き続き必須
    for $pawn in $pawn_list
      if ($target_gender == "ALL" || $pawn.gender == $target_gender) && (find_interaction_line $pawn.fullinteraction $keyword_dict) != ""
        ret $pawn
      end
    end

    ret null
  end

  # ===== 除外/置換キーワード（key: 検索文字列, value: 置換後） =====
  replace_dict = {
    "Male": "男性",
    "Female": "女性",
    "Traits: ": "",
    "Relations: ": "",
    "Equipment: ": "",
    "Memory: ": "",
    "[Weapon]": "武器：",
    "[Apparel]": "衣服：",
    "防弾チョッキ": "",
    "不死": "",
    "不老": "",
    "メカリンク": "",
    "豪華な宿舎": "",
  }

  # ===== 表示する特性（key: 検索文字列, value: 表示名） =====
  # ここに存在しない特性は表示しない
  trait_display_dict = {
    "勤労者": "勤労者",
  }

  # ===== 欲求の表示確率 =====
  # key: 部分一致する文字列
  # value: 表示確率（0～100）
  thought_display_dict = {
    "入植者が死んだ": 1,
  }

  # ===== 文字列内の "(...)" を丸ごと除去する =====
  func strip_parens
    $text = $0
    $parts = $text | string.split "("
    $result = ""
    $first = true
    for $part in $parts
      if $first
        $result = $result + $part
        $first = false
      else
        $inner = $part | string.split ")"
        if $inner.size > 1
          $result = $result + ($inner | array.last)
        end
        # inner.size == 1 の場合（閉じ括弧の後ろに何もない）は何も追加しない
      end
    end
    ret ($result | string.strip)
  end

  # ===== 装備情報から武器部分だけを取得する =====
  # 例: "[Weapon] ナイフ | [Apparel] Ｔシャツ" → "[Weapon] ナイフ"
  func extract_weapon
    $text = $0
    if $text == null || $text == "" || !($text | string.contains "[Weapon]")
      ret ""
    end

    $weapon_parts = $text | string.split "[Weapon]"
    $apparel_parts = $weapon_parts[1] | string.split "[Apparel]"
    $weapon = $apparel_parts[0] | string.replace "|" "" | string.strip

    if $weapon == ""
      ret ""
    end

    ret "[Weapon] " + $weapon
  end

  # ===== 場所情報を一定の形式に整え、天候・外部環境を補う =====
  # 温度は35℃以上または-10℃以下の場合だけ表示し、軌道上では表示しない
  func format_location
    $text = $0
    $weather_text = $1

    if $text == null || $text == ""
      if $weather_text != null && $weather_text != ""
        ret "天候・外部環境：" + $weather_text
      end
      ret ""
    end

    $parts = $text | string.split ";"
    $location_type = $parts[0] | string.strip
    $result = ""
    $is_orbital = $weather_text != null && ($weather_text | string.contains "軌道上")

    if $location_type | string.contains "室内"
      $result = "区分：室内"
    else if $location_type | string.contains "外気"
      $result = "区分：屋外"
    else
      $result = "区分：" + $location_type
    end

    # p.locationの3要素目は部屋名
    if $parts.size > 2
      $room_name = $parts[2] | string.strip
      if $room_name != ""
        $result = $result + " / 部屋：" + $room_name
      end
    end

    # p.locationの2要素目はpawn位置の実温度。会話に影響しやすい極端な時だけ残す
    if $parts.size > 1 && !$is_orbital
      $temperature_text = $parts[1] | string.strip
      if $temperature_text != ""
        $temperature_number_text = $temperature_text | string.replace "℃" "" | string.replace "°" "" | string.replace "C" "" | string.replace "c" "" | string.strip
        $temperature_value = $temperature_number_text | string.to_float
        if $temperature_value != null && ($temperature_value >= 35 || $temperature_value <= -10)
          $result = $result + " / 温度：" + $temperature_text
        end
      end
    end

    if $weather_text != null && $weather_text != ""
      $result = $result + " / 天候・外部環境：" + $weather_text
    end

    ret $result
  end

  # ===== カンマ区切り文字列の後処理（空要素・余分な空白の除去） =====
  func clean_list
    $text = $0
    $parts = $text | string.split ","
    $result = ""
    for $part in $parts
      $trimmed = $part | string.strip
      if $trimmed != ""
        if $result != ""
          $result = $result + ", "
        end
        $result = $result + $trimmed
      end
    end
    ret $result
  end
    # ===== 血縁関係を現在の会話に関係する人物だけに絞る =====
  # 例:
  #   リンダ(妻), カーミラ(姉)
  #
  # 以下のどちらかを満たす場合だけ残す
  # ・名前が pawns 内に存在する
  # ・名前が現在の p.job に含まれる
  func filter_relations
    $text = $0
    $pawn_list = $1
    $job_text = $2

    if $text == null || $text == ""
      ret ""
    end

    $parts = $text | string.split ","
    $result = ""

    for $part in $parts
      $relation = $part | string.strip

      if $relation != ""

        # "リンダ(妻)" → "リンダ"
        $name_parts = $relation | string.split "("
        $relation_name = $name_parts[0] | string.strip

        $keep = false

        # ===== 現在会話中のpawnにいるか =====
        for $other_p in $pawn_list
          if $other_p.name && $relation_name == ($other_p.name | string.strip)
            $keep = true
            break
          end
        end

        # ===== 現在のjobに名前が含まれているか =====
        if !$keep && $job_text && ($job_text | string.contains $relation_name)
          $keep = true
        end

        # ===== 条件を満たした関係だけ追加 =====
        if $keep
          if $result != ""
            $result = $result + ", "
          end
          $result = $result + $relation
        end

      end
    end

    ret $result
  end

  # ===== 部分置換（内部利用）。置換後にclean_listで整形 =====
  # keyが重複・包含関係にある場合は登録順で結果が変わるため、具体的な長いkeyを先に置く
  func partial_replace
    $text = $0
    $dict = $1
    $result = $text
    for $key in ($dict | object.keys)
      $keyword = $dict[$key]
      $result = $result | string.replace $key $keyword
    end
    ret (clean_list $result)
  end

  # ===== 全置換（内部利用） =====
  func whole_replace
    $text = $0
    $dict = $1
    for $key in ($dict | object.keys)
      if $text && ($text | string.contains $key)
        ret $dict[$key]
      end
    end
    ret $text
  end

  # ===== メインの置換窓口 =====
  # mode: "partial" = 部分置換 / "whole" = 全置換 / null等 = 無置換
  func apply_replace
    $value = $0
    $dict = $1
    $mode = $2
    if $value == null || $dict == null
      ret $value
    end
    if $mode == "partial"
      ret (partial_replace $value $dict)
    end
    if $mode == "whole"
      ret (whole_replace $value $dict)
    end
    ret $value
  end

  # ===== 健康情報専用の置換 =====
  # valueが空文字: keyを含むHediff項目全体を除外
  # valueが空以外: keyに一致した部分だけをvalueへ置換
  func replace_health_label
    $health_text = $0
    $health_dict = $1

    if $health_text == null || $health_text == "" || $health_dict == null
      ret $health_text
    end

    # simple function内の通常変数は呼び出し元と共有されるため、$付きのローカル変数を使う
    $health_result = $health_text
    for $health_key in ($health_dict | object.keys)
      if $health_result | string.contains $health_key
        $keyword = $health_dict[$health_key]
        if $keyword == ""
          ret ""
        end
        $health_result = $health_result | string.replace $health_key $keyword
      end
    end

    ret ($health_result | string.strip)
  end

  # ===== RimTalkの健康情報から出血状態を抜き出す =====
  # 例: "Health: Bleeding (severe, death in 19.2 hours), ..."
  #  → "Bleeding (severe, death in 19.2 hours)"
  func extract_bleeding_status
    $text = $0
    $marker = "Bleeding ("

    if $text == null || $text == "" || !($text | string.contains $marker)
      ret ""
    end

    $parts = $text | string.split $marker
    for $part in $parts
      if for.index > 0
        $detail_parts = $part | string.split ")"
        if $detail_parts.size > 0 && $detail_parts[0] != ""
          ret $marker + $detail_parts[0] + ")"
        end
      end
    end

    ret ""
  end

  # ===== 指定時間内に付与された健康状態を取得（除外済みのVisible Hediff） =====
  func format_health_conditions
    $pawn = $0
    $threshold_ticks = $1
    $health_text = $pawn.health
    $result = ""

    # p.healthが空なら、RimTalk側で健康情報が無効になっている
    if $health_text == null || $health_text == ""
      ret ""
    end

    for $hediff in (GetHediffs $pawn)
      if $hediff.ageTicks <= $threshold_ticks
        # 空のvalueは項目全体の除外、それ以外は一致部分の置換に使う
        $health_label = replace_health_label $hediff.LabelCap replace_dict

        if $health_label != ""
          $hours_ago = ($hediff.ageTicks / 2500) | math.round 1
          $entry = $health_label

          # 全身HediffはPartを持たないため、明示的に「全身」と表示する
          if $hediff.Part
            $entry = $entry + " (" + $hediff.Part.Label + ")"
          else
            $entry = $entry + " (全身)"
          end

          $entry = $entry + "(" + $hours_ago + "時間前)"
          if $result != ""
            $result = $result + ", "
          end
          $result = $result + $entry
        end
      end
    end

    # 単なる「出血中」ではなく、重症度と死亡までの予測時間を保持する
    $bleeding_status = extract_bleeding_status $health_text
    if $bleeding_status != ""
      if $result != ""
        $result = $bleeding_status + ", " + $result
      else
        $result = $bleeding_status
      end
    end

    ret $result
  end

  # ===== 辞書に登録された特性だけを表示する =====
  # keyは特性名への部分一致、valueは実際に表示する名前
  func filter_traits
    $trait_text = $0
    $trait_dict = $1
    $trait_result = ""

    if $trait_text == null || $trait_text == "" || $trait_dict == null
      ret ""
    end

    for $trait in ($trait_text | string.split ",")
      $trimmed_trait = $trait | string.strip
      for $trait_key in ($trait_dict | object.keys)
        if $trimmed_trait | string.contains $trait_key
          $display_trait = $trait_dict[$trait_key] | string.strip
          if $display_trait != ""
            if $trait_result != ""
              $trait_result = $trait_result + ", "
            end
            $trait_result = $trait_result + $display_trait
          end
          break
        end
      end
    end

    ret $trait_result
  end

  # ===== ラベル+値の1ブロックを作る。値がnull/空文字ならスキップ =====
  func add_line
    $label = $0
    $value = $1
    if $value != null && $value != ""
      ret "【" + $label + "】\n" + $value + "\n"
    end
    ret ""
  end

  # ===== フィールド1つ分を「危険度判定→置換→整形」まで一括処理 =====
  # danger_flag: true/false = 判定する / null = 判定しない（常に表示扱い）
  # dict, mode: 置換したい場合に指定（不要ならdict=null）
  func prepare_field
    $label = $0
    $value = $1
    $danger = $2
    $dict = $3
    $mode = $4
    
    if $danger != null && !$danger
      ret ""
    end

    $value = apply_replace $value $dict $mode
    ret (add_line $label $value)
  end

  # ===== 疑似乱数 =====
  func simple_random
    $seed_num = $0
    $value = ($seed_num * 9301) + 49297
    $value = $value % 233280
    if $value < 0
      $value = $value + 233280
    end
    ret $value
  end

  # 0〜99の数字を返す（%判定に使う）
  func random_percent
    $seed_num = $0
    ret (simple_random $seed_num) % 100
  end

  # ===== 欲求を表示確率に従って抽出 =====
  func filter_thoughts
    $text = $0
    $dict = $1
    $pawn_seed = $2
    $current_tick = $3

    if $text == null || $text == "" || $dict == null
      ret ""
    end

    $parts = $text | string.split ","
    $result = ""

    for $part in $parts
      $trimmed = $part | string.strip
      $show = false

      # Tickを混ぜることで、呼び出すタイミングによって結果が変わる
      $thought_seed = $current_tick + $pawn_seed + (for.index * 97) + ($trimmed | string.size)

      for $key in ($dict | object.keys)
        if $trimmed != "" && ($trimmed | string.contains $key)

          $chance = $dict[$key]

          if $chance >= 100
            $show = true
          else
            $dice = random_percent $thought_seed

            if $dice < $chance
              $show = true
            end
          end

          break
        end
      end

      if $show
        if $result != ""
          $result = $result + ", "
        end
        $result = $result + $trimmed
      end
    end

    ret $result
  end

  # ===== personalityに特殊文字列が含まれていた場合の処理 =====
  # text: p.personalityそのもの
  # context_text: {"キーワード"}判定に使う参照テキスト（例: 種族名や説明文など）
  # danger_flag: {!}判定に使うtrue/false
  # pawn_seed: このpawn専用の乱数シード
  # NS_text: fullinteractionから取得・置換したNS文字列。{NS}判定にも使う
  # 制約: 特殊構文は1行につき1個とし、本文中には追加の「}」を含めない
  # 通常行に波括弧を使うと特殊構文と解釈されるため、人物設定では避ける
  func check_personality
    $text = $0
    $context_text = $1
    $danger = $2
    $pawn_seed = $3
    $NS_text = $4

    $split_text = $text | string.split "\n"
    $result = ""

    for $line in $split_text

      if ($line | string.contains "{") && ($line | string.contains "}")

        # ---- 1. "}" で2つに割る ----
        # 例: {20%}実は趣味がある{END}
        #  → parts[0] = "{20%"
        #  → parts[1] = "実は趣味がある{END"
        $parts = $line | string.split "}"

        if $parts.size >= 2

          $tag_part  = $parts[0]
          $body_part = $parts[1]

          # ---- 2. タグの中身だけ取り出す（先頭の "{" を削除） ----
          $tag = $tag_part | string.replace "{" ""

          # ---- 3. 本文だけ取り出す（末尾の "{END" を削除） ----
          $body = $body_part | string.replace "{END" ""

          $show_line = false

          if $tag | string.contains "%"
            # ===== パターンA: {20%}あいうえお{END} =====
            # "20%" → "20" にしてから数値に変換
            $percent_num = ($tag | string.replace "%" "") | string.to_int

            # ポーンごと・行ごとに違う乱数になるよう、for.indexも混ぜる
            $line_seed = $pawn_seed + for.index
            $dice = random_percent $line_seed

            if $dice < $percent_num
              $show_line = true
            end

          else
            if $tag == "!"
              # ===== パターンB: {!}あいうえお{END} =====
              if $danger
                $show_line = true
              end
            else if $tag == "NS"
              # ===== パターンC: {NS}あいうえお{END} =====
              # fullinteractionからNS文字列を取得できた場合だけ表示する
              if $NS_text != null && $NS_text != ""
                $show_line = true
              end
            else
              # ===== パターンD: {"キーワード"}あいうえお{END} =====
              $keyword = $tag | string.replace "\"" ""
              if $context_text && ($context_text | string.contains $keyword)
                $show_line = true
              end
            end
          end

          if $show_line
            if $result != ""
              $result = $result + "\n"
            end
            $result = $result + $body
          end

        end

      else
        # 特殊記法が含まれていない普通の行はそのまま出力
        if $result != ""
          $result = $result + "\n"
        end
        $result = $result + $line
      end

    end

    ret $result
  end
